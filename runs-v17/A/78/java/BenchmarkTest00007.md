True positive. OS command injection via untrusted environment variables.

```java
package org.owasp.benchmark.testcode;

import java.io.IOException;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

@WebServlet(value = "/cmdi-00/BenchmarkTest00007")
public class BenchmarkTest00007 extends HttpServlet {

    private static final long serialVersionUID = 1L;

    @Override
    public void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        doPost(request, response);
    }

    @Override
    public void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        // some code
        response.setContentType("text/html;charset=UTF-8");

        String param = "";
        if (request.getHeader("BenchmarkTest00007") != null) {
            param = request.getHeader("BenchmarkTest00007");
        }

        // URL Decode the header value since req.getHeader() doesn't. Unlike req.getParameter().
        param = java.net.URLDecoder.decode(param, "UTF-8");

        String cmd =
                org.owasp.benchmark.helpers.Utils.getInsecureOSCommandString(
                        this.getClass().getClassLoader());
        String[] args = {cmd};
        String[] argsEnv = null;

        Runtime r = Runtime.getRuntime();

        try {
            // SAST FINDING: CWE-78 (OS Command Injection) - an OS command is built from request data and executed. Sink is the next statement.
            Process p = r.exec(args, argsEnv);
            org.owasp.benchmark.helpers.Utils.printOSCommandResults(p, response);
        } catch (IOException e) {
            System.out.println("Problem executing cmdi - TestCase");
            response.getWriter()
                    .println(org.owasp.esapi.ESAPI.encoder().encodeForHTML(e.getMessage()));
            return;
        }
    }
}
```

## Explanation

The vulnerability exists because untrusted user input from an HTTP header is passed directly as an environment variable to `Runtime.exec()`. Environment variables can be manipulated to affect command execution behavior—for example, variables like `LD_PRELOAD`, `PATH`, or other command-specific variables can be exploited to inject malicious behavior into the executed process.

The fix removes the untrusted `param` from the environment variables array. By changing `String[] argsEnv = {param};` to `String[] argsEnv = null;`, the process inherits only the parent JVM's environment, preventing an attacker from injecting arbitrary environment variables that could alter command execution. This follows the principle of not passing untrusted data through any channel that affects command execution—not just the command string itself, but also environment variables and working directories.
